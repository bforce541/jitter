cmake_minimum_required(VERSION 3.20)
project(jitter VERSION 1.0.0 LANGUAGES C CXX)

set(JITTER_VARIANT "any" CACHE STRING "Restrict engine to one registered implementation")
set_property(CACHE JITTER_VARIANT PROPERTY STRINGS any coarse narrow map ordered scan indexed)
if(NOT JITTER_VARIANT MATCHES "^(any|coarse|narrow|map|ordered|scan|indexed)$")
    message(FATAL_ERROR "Unsupported JITTER_VARIANT: ${JITTER_VARIANT}")
endif()
message(STATUS "Jitter compiled variant: ${JITTER_VARIANT}")
file(SHA256 "${CMAKE_SOURCE_DIR}/third_party/nlohmann/json.hpp" JITTER_JSON_SHA256)
if(NOT JITTER_JSON_SHA256 STREQUAL "aaf127c04cb31c406e5b04a63f1ae89369fccde6d8fa7cdda1ed4f32dfc5de63")
    message(FATAL_ERROR "Vendored nlohmann/json header does not match pinned upstream release")
endif()
message(STATUS "nlohmann/json 3.12.0: ${JITTER_JSON_SHA256}")
file(SHA256 "${CMAKE_SOURCE_DIR}/third_party/sqlite/sqlite3.c" JITTER_SQLITE_SHA256)
file(SHA256 "${CMAKE_SOURCE_DIR}/third_party/sqlite/sqlite3.h" JITTER_SQLITE_HEADER_SHA256)
if(NOT JITTER_SQLITE_SHA256 STREQUAL "6c35bc5f7f85eac9c49928bacbb02bb694b547aabf69197e058cca245ad80e83" OR NOT JITTER_SQLITE_HEADER_SHA256 STREQUAL "89b62c671c5964e137409ce034941b7b05a3af2c9875aba41f47f9483d0c2515")
    message(FATAL_ERROR "Vendored SQLite files do not match the pinned release")
endif()

if(NOT APPLE AND NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "Phase 1 resource collection supports Linux and macOS")
endif()
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()

if(CMAKE_CONFIGURATION_TYPES)
    message(FATAL_ERROR "Phase 1 requires a single-configuration CMake generator")
endif()

find_package(Threads REQUIRED)
find_package(Git QUIET)
set(JITTER_SOURCE_REVISION "unknown")
set(JITTER_SOURCE_DIRTY "true")
if(Git_FOUND AND EXISTS "${CMAKE_SOURCE_DIR}/.git")
    execute_process(COMMAND "${GIT_EXECUTABLE}" rev-parse HEAD
        WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}" OUTPUT_VARIABLE JITTER_SOURCE_REVISION OUTPUT_STRIP_TRAILING_WHITESPACE)
    execute_process(COMMAND "${GIT_EXECUTABLE}" status --porcelain
        WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}" OUTPUT_VARIABLE JITTER_GIT_STATUS OUTPUT_STRIP_TRAILING_WHITESPACE)
    if(JITTER_GIT_STATUS STREQUAL "")
        set(JITTER_SOURCE_DIRTY "false")
    endif()
endif()
string(TOUPPER "${CMAKE_BUILD_TYPE}" JITTER_BUILD_TYPE_UPPER)
set(JITTER_INFO_FLAGS "${CMAKE_CXX_FLAGS} ${CMAKE_CXX_FLAGS_${JITTER_BUILD_TYPE_UPPER}} -Wall -Wextra -Wpedantic; C++20; extensions=off")
string(REPLACE "\\" "\\\\" JITTER_INFO_FLAGS "${JITTER_INFO_FLAGS}")
string(REPLACE "\"" "\\\"" JITTER_INFO_FLAGS "${JITTER_INFO_FLAGS}")
configure_file(include/jitter/build_info.hpp.in generated/jitter/build_info.hpp @ONLY)

add_library(jitter_sqlite STATIC third_party/sqlite/sqlite3.c)
set_target_properties(jitter_sqlite PROPERTIES C_STANDARD 99)
target_compile_definitions(jitter_sqlite PRIVATE SQLITE_THREADSAFE=1 SQLITE_OMIT_LOAD_EXTENSION=1 SQLITE_DQS=0)
target_include_directories(jitter_sqlite PUBLIC third_party/sqlite)
target_link_libraries(jitter_sqlite PUBLIC Threads::Threads ${CMAKE_DL_LIBS})
add_library(jitter_core src/application.cpp src/experiment.cpp src/trace.cpp src/sqlite_dataset.cpp)
target_include_directories(jitter_core PUBLIC include)
target_include_directories(jitter_core SYSTEM PRIVATE third_party)
target_compile_definitions(jitter_core PRIVATE JITTER_COMPILED_VARIANT="${JITTER_VARIANT}")
target_compile_features(jitter_core PUBLIC cxx_std_20)
set_target_properties(jitter_core PROPERTIES CXX_EXTENSIONS OFF)
target_compile_options(jitter_core PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(jitter_core PUBLIC Threads::Threads jitter_sqlite)

add_executable(jitter src/main.cpp)
target_include_directories(jitter PRIVATE "${CMAKE_BINARY_DIR}/generated")
target_link_libraries(jitter PRIVATE jitter_core)
set_target_properties(jitter PROPERTIES CXX_EXTENSIONS OFF)
target_compile_options(jitter PRIVATE -Wall -Wextra -Wpedantic)

include(CTest)
if(BUILD_TESTING)
    add_executable(jitter_core_tests tests/core_tests.cpp)
    target_link_libraries(jitter_core_tests PRIVATE jitter_core)
    set_target_properties(jitter_core_tests PROPERTIES CXX_EXTENSIONS OFF)
    add_test(NAME core COMMAND jitter_core_tests)
    find_package(Python3 3.9 REQUIRED COMPONENTS Interpreter)
    add_test(NAME foundation COMMAND "${CMAKE_COMMAND}" -E env
        "JITTER_TEST_BINARY=$<TARGET_FILE:jitter>"
        "${Python3_EXECUTABLE}" -m unittest discover
        -s "${CMAKE_SOURCE_DIR}/tests" -p "test_*.py" -v)
    set_tests_properties(core foundation PROPERTIES TIMEOUT 180)
endif()
